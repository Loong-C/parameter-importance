\begin{frame}{Stage 1｜我们到底在估计什么？}
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{block}{固定参数状态下的局部梯度空间贡献}
        \[
          \widehat C^{U}_{k,t}
          =\eta_{g(k),t}\,
          \frac{S_{1,k,t}^{2}-S_{2,k,t}}{M(M-1)}
        \]
      \end{block}
      \begin{itemize}
        \item $S_{1,k,t}$：$M$ 个 microbatch 梯度之和
        \item $S_{2,k,t}$：对应梯度平方和
        \item 去除对角项：目标指向总体梯度平方
      \end{itemize}
    \column{0.42\textwidth}
      \centering
      \begin{tikzpicture}[node distance=0.62cm]
        \node[flowbox, text width=3.1cm] (mb) {$M$ 个 microbatch\\梯度 $g_{1,k},\ldots,g_{M,k}$};
        \node[flowbox, text width=3.1cm, below=of mb] (ss) {充分统计量\\$S_1,\ S_2$};
        \node[greenbox, text width=3.1cm, below=of ss] (u) {去对角 U-statistic\\$\widehat C^U_{k,t}$};
        \draw[arr] (mb) -- (ss);
        \draw[strongarr] (ss) -- (u);
      \end{tikzpicture}
      \vspace{5pt}
      {\small 与参数绝对值、移动量、raw 分数并列比较}
  \end{columns}
  \keyline{这一阶段只核对实现是否忠实；模型机理留到后面讨论。}
  \note{
    这里固定参数状态 $\Theta_t$，只看同一时刻的梯度统计。$M$ 是 microbatch 数；$S_{1,k}$ 是第 $k$ 个参数在所有 microbatch 上的梯度和，$S_{2,k}$ 是这些梯度的平方和。减去 $S_2$ 就是在去掉每个梯度与自身相乘的对角项。\par
    $\eta_{g(k),t}$ 是参数所在优化器组的学习率。这个局部分数和 AdamW 的完整路径贡献不是同一个量。单步 U-statistic 允许为负；负值来自有限样本波动，若先截断，再讨论无偏性就失去了意义。
  }
\end{frame}

\begin{frame}{Stage 1｜double 与 U 怎样消掉同源偏差？}
  \textbf{偏差来自对角项：同一个随机梯度与自己相乘，会额外带入方差。}
  \vspace{3pt}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{double sample｜换成两组独立梯度}
        \[
          \begin{aligned}
            \widehat{\mu_k^2}^{\,\mathrm{double}}
              &=\bar g_k^{(A)}\bar g_k^{(B)},\\
            \mathbb E[\widehat{\mu_k^2}^{\,\mathrm{double}}]
              &=\mu_k^2
          \end{aligned}
        \]
      \end{block}
      \begin{itemize}
        \item A、B 独立：乘积期望可拆开
        \item 代价：需要两组独立梯度
      \end{itemize}
    \column{0.48\textwidth}
      \begin{block}{U-statistic｜在同一梯度池中删掉对角项}
        \[
          \begin{aligned}
            \widehat{\mu_k^2}^{\,U}
              &=\frac{\sum_{m\ne n}g_{m,k}g_{n,k}}{M(M-1)},\\
              &=\frac{S_{1,k}^2-S_{2,k}}{M(M-1)}
          \end{aligned}
        \]
      \end{block}
      \begin{itemize}
        \item $S_1^2$ 含全部两两乘积，$S_2$ 正好是自乘项
        \item 复用同一梯度池；单次估计可能为负
      \end{itemize}
  \end{columns}
  \vspace{3pt}
  \centering
  \begin{tikzpicture}[node distance=0.85cm]
    \node[warmbox, text width=3.2cm] (raw) {raw\\同源梯度自乘};
    \node[greenbox, text width=3.5cm, right=of raw] (double) {double\\两组独立梯度交叉相乘};
    \node[greenbox, text width=3.5cm, right=of double] (u) {U-statistic\\删除 $m=n$ 自乘项};
    \draw[arr] (raw) -- (double);
    \draw[arr] (double) -- (u);
  \end{tikzpicture}
  \keyline{两种方法都把“随机梯度自乘”换成“独立梯度交叉相乘”。}
  \note{
    raw 使用同一批梯度的平方，因此包含 $m=n$ 的对角项；这些自乘项的期望是 $\mu_k^2+\sigma_k^2$，方差由此进入目标。\par
    double sample 把样本分成独立的 A、B 两组。独立性使乘积的期望可以拆成 $\mathbb E[\bar g^{(A)}]\mathbb E[\bar g^{(B)}]=\mu_k^2$。U-statistic 不另取第二组数据，而是在同一组 $M$ 个 microbatch 梯度中删掉所有 $m=n$ 的自乘项，只平均 $m\ne n$ 的交叉项。\par
    两者都依赖样本独立或条件独立。double 更直观但需要两组梯度；U 更节省样本，却要维护 $S_1$、$S_2$，且单次估计可能为负。下一页再看 raw 偏差为什么按 $1/B$ 衰减。
  }
\end{frame}

% 07
\begin{frame}{Stage 2｜raw 为什么会系统性过估计？}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{同一批数据同时提供两个梯度因子}
      \[
        \mathbb E[\bar g_k^2]
        =\mu_k^2+\frac{\sigma_k^2}{B}
      \]
      \begin{itemize}
        \item 目标：$\mu_k^2$
        \item 额外项：同源随机梯度方差
        \item 预测：正偏随 batch size 按 $1/B$ 衰减
      \end{itemize}
    \column{0.46\textwidth}
      \centering
      \begin{tikzpicture}[x=0.90cm,y=0.90cm]
        \draw[-{Stealth}, thick] (0,0) -- (5.7,0) node[right, font=\small] {$1/B$};
        \draw[-{Stealth}, thick] (0,0) -- (0,4.3) node[above, font=\small] {期望分数};
        \draw[neutral, thick, dashed] (0,0.9) -- (5.2,0.9);
        \draw[positive, thick] plot[domain=0:5.1] (\x,{0.9+0.50*\x});
        \foreach \x in {0.6,1.8,3.0,4.2,5.0} {
          \pgfmathsetmacro{\yraw}{0.9+0.50*\x}
          \fill[positive] (\x,\yraw) circle (2.5pt);
        }
        \node[font=\small, text=neutral, anchor=west] at (3.5,0.55) {$\mu_k^2$};
        \node[font=\small, text=positive, anchor=west] at (3.25,3.35) {raw（示意）};
      \end{tikzpicture}
      \vspace{3pt}
      {\small 图示表达预注册理论趋势，不代表已有结果。}
  \end{columns}
  \keyline{Stage 2 先把梯度噪声带来的偏差从真实信号里剥离出来。}
  \note{
    设单样本梯度的均值为 $\mu_k$、方差为 $\sigma_k^2$，batch 中有 $B$ 个独立样本。batch 平均梯度的方差是 $\sigma_k^2/B$，因此它的平方期望比目标 $\mu_k^2$ 多出一个正项。batch 越大，这个偏差越小，但不会因为重复实验而自动消失。\par
    图上横轴使用 $1/B$，理论预测 raw 应接近一条线性趋势。这里固定参数状态，只测估计量，不执行 optimizer step，以免参数更新又引入新的变化。
  }
\end{frame}

\begin{frame}{Stage 3｜沿实际更新端点定义路径贡献}
  \begin{columns}[T]
    \column{0.53\textwidth}
      \textbf{固定更新前后端点与独立 probe 损失}
      \begin{align*}
        \Gamma_t(\alpha)&=\Theta_t+\alpha\Delta\Theta_t,\\
        C^{\mathrm{path},\mathcal P}_{k,t}
        &=-\Delta\theta_{k,t}\int_0^1
          \partial_k\mathcal L_{\mathcal P}(\Gamma_t(\alpha))\,d\alpha.
      \end{align*}
      \begin{itemize}
        \item 同一线性路径、同一确定性 probe
        \item 实际端点位移已包含优化器作用
        \item 不再额外乘学习率或裁剪因子
      \end{itemize}
    \column{0.43\textwidth}
      \centering
      \begin{tikzpicture}[x=1cm,y=1cm]
        \draw[strongarr] (0,1.4) -- (5.2,1.4);
        \fill[positive] (0.35,1.4) circle (3.5pt);
        \fill[emphasis] (4.85,1.4) circle (3.5pt);
        \node[above=6pt, font=\small] at (0.35,1.4) {$\Theta_t$};
        \node[above=6pt, font=\small] at (4.85,1.4) {$\Theta_{t+1}$};
        \foreach \x/\a in {1.25/0.2,2.60/0.5,3.95/0.8} {
          \draw[negative, thick] (\x,1.18) -- (\x,1.62);
          \node[below=5pt, font=\footnotesize, text=negative] at (\x,1.18) {$\alpha=\a$};
        }
        \draw[decorate, decoration={brace, amplitude=6pt, mirror}, thick, neutral]
          (0.35,0.45) -- (4.85,0.45) node[midway, below=9pt, font=\small] {同一条参数路径};
      \end{tikzpicture}
      \begin{exampleblock}{完备性检查}
        $\sum_k C_{k,t}=\mathcal L_{\mathcal P}(\Theta_t)-\mathcal L_{\mathcal P}(\Theta_{t+1})$
      \end{exampleblock}
  \end{columns}
  \note{
    $\Theta_t$ 和 $\Theta_{t+1}$ 是一次真实优化更新前后的参数，$\Delta\Theta_t$ 已经包含学习率、动量、权重衰减和裁剪等优化器作用，因此路径贡献里不能再次乘这些因子。\par
    probe 是一批固定、独立且不参与这次更新的数据，用它测量路径上的损失梯度。完备性表示所有参数贡献求和，应恢复两个端点之间的 probe loss 差。probe 的抽样波动与求积截断误差会分开报告。
  }
\end{frame}

\begin{frame}{备份｜raw 正偏从哪里来？}
  \textbf{为什么同一个 minibatch 的平均梯度平方不是 $\mu_k^2$？}
  \begin{align*}
    \bar g_k &= \frac{1}{B}\sum_{i=1}^{B} g_k(z_i),
    &\mathbb E[\bar g_k]&=\mu_k,\\
    \mathbb E[\bar g_k^2]
      &=\operatorname{Var}(\bar g_k)+\mathbb E[\bar g_k]^2,
    &\operatorname{Var}(\bar g_k)&=\frac{\sigma_k^2}{B},\\
    \Rightarrow\quad
    \mathbb E[\bar g_k^2]&=\mu_k^2+\frac{\sigma_k^2}{B}.&&
  \end{align*}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{exampleblock}{标量例子}
        \[
          \mu=2,\quad \sigma^2=9,\quad B=9
          \;\Rightarrow\; \mathbb E[\bar g^2]=5
        \]
      \end{exampleblock}
    \column{0.48\textwidth}
      \begin{block}{判别预测}
        \begin{itemize}
          \item 真目标：$\mu^2=4$
          \item raw 额外增加 1
          \item $B$ 增大时偏差按 $1/B$ 衰减
        \end{itemize}
      \end{block}
  \end{columns}
  \keyline{double sample 和 U-statistic 都是在去掉同源对角项。}
  \note{
    $\bar g_k$ 是 $B$ 个样本梯度的平均。它的期望是总体均值 $\mu_k$，方差是 $\sigma_k^2/B$。利用 $\mathbb E[X^2]=\operatorname{Var}(X)+\mathbb E[X]^2$，就得到 raw 比目标 $\mu_k^2$ 多出 $\sigma_k^2/B$。\par
    例子中目标是 4，raw 的期望是 5，多出的 1 正是方差项。若样本相关，$\operatorname{Var}(\bar g)$ 还包含协方差，不能继续使用独立样本下的简式。
  }
\end{frame}

\begin{frame}{备份｜等权与加权 U-statistic}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{等权 microbatch}
      \[
        \widehat{\mu_k^2}^{\,U}
        =\frac{S_{1,k}^2-S_{2,k}}{M(M-1)}
        =\frac{1}{M(M-1)}\sum_{m\ne n}g_{m,k}g_{n,k}
      \]
      \begin{itemize}
        \item $S_{1,k}=\sum_m g_{m,k}$
        \item $S_{2,k}=\sum_m g_{m,k}^2$
        \item $M\ge2$；交换顺序结果不变
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{不等有效 token 数}
      \[
        \widehat{\mu_k^2}^{\,U_w}
        =\frac{G_{1,k}^2-G_{2,k}}{N_1^2-N_2}
      \]
      \begin{itemize}
        \item $G_{1,k}=\sum_m b_m g_{m,k}$
        \item $G_{2,k}=\sum_m b_m^2 g_{m,k}^2$
        \item $N_1=\sum_m b_m$，$N_2=\sum_m b_m^2$
      \end{itemize}
  \end{columns}
  \vspace{6pt}
  \begin{exampleblock}{一致性检查}
    当所有 $b_m$ 相等时，加权式退化为等权式；分母必须严格为正。
  \end{exampleblock}
  \keyline{权重应与训练损失的统计单位一致；microbatch 不能默认等权。}
  \note{
    等权式用 $M(M-1)$ 个有序交叉对，$S_1^2-S_2$ 是它的快速计算形式。若各 microbatch 的有效 token 数 $b_m$ 不同，梯度在训练损失中的权重也不同，因此要改用 $G_1$、$G_2$、$N_1$、$N_2$ 的加权式。\par
    分母 $N_1^2-N_2$ 是所有不同 microbatch 权重乘积之和，必须严格为正。所有 $b_m$ 相等时，加权式应退化为等权式，这是重要的一致性检查。
  }
\end{frame}

\begin{frame}{备份｜路径积分的完备性从哪里来？}
  \textbf{固定线性路径 $\Gamma(\alpha)=\Theta_0+\alpha\Delta\Theta$ 与标量损失 $\mathcal L$。}
  \begin{align*}
    \frac{d}{d\alpha}\mathcal L(\Gamma(\alpha))
      &=\nabla_\Theta \mathcal L(\Gamma(\alpha))^\top\Delta\Theta,\\
    \mathcal L(\Theta_1)-\mathcal L(\Theta_0)
      &=\int_0^1\sum_k
        \partial_k\mathcal L(\Gamma(\alpha))\,\Delta\theta_k\,d\alpha.
  \end{align*}
  定义 $C_k^{\mathrm{path}}=-\Delta\theta_k\int_0^1\partial_k\mathcal L(\Gamma(\alpha))d\alpha$，则
  \[
    \boxed{\sum_k C_k^{\mathrm{path}}
      =\mathcal L(\Theta_0)-\mathcal L(\Theta_1)}
  \]
  \begin{center}
  \begin{tikzpicture}[node distance=0.65cm]
    \node[flowbox, text width=2.8cm] (path) {固定端点与路径};
    \node[greenbox, text width=2.8cm, right=of path] (grad) {沿路径积分梯度};
    \node[purplebox, text width=2.8cm, right=of grad] (sum) {逐参数贡献求和};
    \node[warmbox, text width=2.8cm, right=of sum] (loss) {得到端点 loss 差};
    \draw[arr] (path) -- (grad);
    \draw[arr] (grad) -- (sum);
    \draw[strongarr] (sum) -- (loss);
  \end{tikzpicture}
  \end{center}
  \note{
    在线性路径上，损失对路径参数 $\alpha$ 的导数等于梯度与端点位移的内积。对 $\alpha$ 从 0 积到 1，根据微积分基本定理得到端点 loss 差；再按参数坐标拆开内积，就得到逐参数贡献。\par
    因此完备性是定义本身的恒等式。数值结果若偏离，应依次检查积分节点是否足够、模型状态和 probe 是否固定，以及浮点精度是否造成累计误差。
  }
\end{frame}
